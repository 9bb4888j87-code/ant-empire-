function treePixel020(id){

 const sg=source.getContext('2d',{willReadFrequently:true});

 const im=sg.getImageData(0,0,W,H),d=im.data,marked=new Uint8Array(W*H),stack=[];
 function pale(i){let r=d[i],g=d[i+1],b=d[i+2],sat=Math.max(r,g,b)-Math.min(r,g,b),v=(r+g+b)/3;
  return d[i+3]<20||(v>181&&sat<38)||(v>220&&sat<70);}
 function add(x,y){if(x<0||y<0||x>=W||y>=H)return;let idx=y*W+x,i=idx*4;
  if(marked[idx]||!pale(i))return;marked[idx]=1;stack.push(idx);}
 for(let x=0;x<W;x++){add(x,0);add(x,H-1)}
 for(let y=0;y<H;y++){add(0,y);add(W-1,y)}
 while(stack.length){let idx=stack.pop(),x=idx%W,y=idx/W|0;d[idx*4+3]=0;
  add(x-1,y);add(x+1,y);add(x,y-1);add(x,y+1);}
 for(let pass=0;pass<2;pass++){
  const toClear=[];
  for(let y=1;y<H-1;y++)for(let x=1;x<W-1;x++){
   const i=(y*W+x)*4,r=d[i],g=d[i+1],b=d[i+2];
   if(!d[i+3]||(r+g+b)/3<151||Math.max(r,g,b)-Math.min(r,g,b)>62)continue;
   if(!d[i-1]||!d[i+7]||!d[i+3-W*4]||!d[i+3+W*4])toClear.push(i);
  }
  for(const i of toClear)d[i+3]=0;
 }
 for(let i=0;i<d.length;i+=4)d[i+3]=d[i+3]<100?0:255;
 sg.putImageData(im,0,0);
 const out=document.createElement('canvas');out.width=59;out.height=69;
 const g=out.getContext('2d',{willReadFrequently:true});g.imageSmoothingEnabled=false;
 g.drawImage(source,0,0,W,H,0,0,59,69);
 const pix=g.getImageData(0,0,59,69),p=pix.data;
 for(let i=0;i<p.length;i+=4){
  if(p[i+3]<120){p[i+3]=0;continue}
  p[i+3]=255;
  for(let k=0;k<3;k++)p[i+k]=Math.min(255,Math.round(p[i+k]/24)*24);
 }
 const old=new Uint8ClampedArray(p);
 for(let y=1;y<68;y++)for(let x=1;x<58;x++){
  const i=(y*59+x)*4;if(!old[i+3])continue;
  if(!old[i-1]||!old[i+7]||!old[i+3-236]||!old[i+3+236])
   for(let k=0;k<3;k++)p[i+k]=Math.floor(p[i+k]*.68);
 }
 g.putImageData(pix,0,0);
 treeCache020[id]=out;return out;
}
